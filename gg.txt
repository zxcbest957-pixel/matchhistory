--[[
    Standalone ViewMatchHistory (BedWars)
    Повністю незалежний скрипт без прив'язки до CatVape чи KingVape.
    Можна запускати окремо в екзекуторі або вбудовувати функцію у власні скрипти.
--]]

local cloneref = cloneref or function(obj) return obj end
local Players = cloneref(game:GetService("Players"))
local ReplicatedStorage = cloneref(game:GetService("ReplicatedStorage"))
local lplr = Players.LocalPlayer

local function viewMatchHistory(targetPlayer)
    targetPlayer = targetPlayer or lplr

    -- 1. Отримання Flamework
    local flameworkSuccess, Flamework = pcall(function()
        return require(ReplicatedStorage:WaitForChild("rbxts_include"):WaitForChild("node_modules"):WaitForChild("@flamework"):WaitForChild("core"):WaitForChild("out")).Flamework
    end)
    if not flameworkSuccess or not Flamework then
        warn("[ViewMatchHistory] Не вдалося завантажити Flamework")
        return false
    end

    -- 2. Отримання AppController (менеджер вікон BedWars)
    local AppController
    pcall(function()
        AppController = Flamework.resolveDependency("@easy-games/game-core:client/controllers/app-controller@AppController")
            or Flamework.resolveDependency("easy-games/game-core:client/controllers/app-controller@AppController")
    end)

    if not AppController then
        pcall(function()
            local AppControllerClass = require(ReplicatedStorage:WaitForChild("rbxts_include"):WaitForChild("node_modules"):WaitForChild("@easy-games"):WaitForChild("game-core"):WaitForChild("out"):WaitForChild("client"):WaitForChild("controllers"):WaitForChild("app-controller")).AppController
            AppController = Flamework.resolveDependency(AppControllerClass)
        end)
    end

    if not AppController then
        warn("[ViewMatchHistory] Не вдалося знайти AppController")
        return false
    end

    -- 3. Отримання модераторського UI компонента MatchHistoryModerationApp
    local moderationAppSuccess, ModerationApp = pcall(function()
        return require(lplr:WaitForChild("PlayerScripts"):WaitForChild("TS"):WaitForChild("controllers"):WaitForChild("global"):WaitForChild("match-history"):WaitForChild("ui"):WaitForChild("match-history-moderation-app")).MatchHistoryModerationApp
    end)

    if not moderationAppSuccess or not ModerationApp then
        warn("[ViewMatchHistory] Не вдалося завантажити MatchHistoryModerationApp")
        return false
    end

    -- 4. Відкриття вікна історії матчів
    local openSuccess, err = pcall(function()
        AppController:openApp({
            appId = "MatchHistoryApp",
            app = ModerationApp
        }, {
            player = targetPlayer,
            matchHistory = {}
        })
    end)

    if openSuccess then
        print("[ViewMatchHistory] Вікно MatchHistory успішно відкрито!")
        return true
    else
        warn("[ViewMatchHistory] Помилка при виклику openApp: " .. tostring(err))
        return false
    end
end

-- Автоматичний виклик при запуску скрипта
viewMatchHistory()

-- Повертаємо функцію, якщо скрипт імпортується через loadstring
return viewMatchHistory
